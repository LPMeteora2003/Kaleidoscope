using Kaleidoscope.Core.Abstractions;
using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace Kaleidoscope.Core
{
    public class YRSingleton<T> : IYRSingleton<T> where T : YRSingleton<T>
    {
        private static readonly Lazy<T> _instance = new(Create);

        public static T Instance => _instance.Value;

        protected YRSingleton() { }

        private static T Create()
        {
            var type = typeof(T);
            var created = Activator.CreateInstance(type, nonPublic: true);
            if (created is not T instance)
                throw new InvalidOperationException($"{type.FullName} must have a non-public parameterless constructor.");
            return instance;
        }
    }
}

using Kaleidoscope.Logic.Core.Abstractions;
using Kaleidoscope.Logic.Core.Abstractions.Evaluation;

namespace Kaleidoscope.Logic.Beaver.SimpleBeaver
{
    public abstract class YRSimpleBeaverBase : IYRLogicEvaluator
    {
        public object Evaluate(IYRLogicConstruct logicConstruct, IYRContext conclusion)
        {
            if (ViolatingJudgement(logicConstruct, conclusion))
                return false;

            if (ComfirmingJudgement(logicConstruct, conclusion))
                return true;

            return null;
        }
        private bool ComfirmingJudgement(IYRLogicConstruct logicConstruct, IYRContext conclusion)
        {
            var judgement = logicConstruct.Judgement;

            if (CompareContext(judgement.Premise, conclusion) && logicConstruct.Verdict == judgement.Verdict)
                return true;

            while (judgement != judgement.Judgement)
            {
                var premise = judgement.Premise;
                var comparisonResult = CompareContext(premise, conclusion);

                if (comparisonResult && logicConstruct.Verdict == judgement.Verdict)
                    return true;
                judgement = judgement.Judgement;
            }

            return false;
        }

        private bool ViolatingJudgement(IYRLogicConstruct logicConstruct, IYRContext conclusion)
        {
            var judgement = logicConstruct.Judgement;

            if (!(CompareContext(judgement.Premise, conclusion) ^ logicConstruct.Verdict != judgement.Verdict))
                return true;

            while (judgement != judgement.Judgement)
            {
                var premise = judgement.Premise;
                var comparisonResult = CompareContext(premise, conclusion);

                if (!(comparisonResult ^ logicConstruct.Verdict != judgement.Verdict))
                    return true;
                judgement = judgement.Judgement;
            }

            return false;
        }


        public abstract bool CompareContext(IYRContext premise, IYRContext conclusion);
    }
}

namespace Kaleidoscope.Logic.Beaver.SimpleBeaver
{
    public class YRSimpleOracleBeaver : YRSimpleBeaverBase, IYRLogicEvaluator
    {
        private readonly IYREqualityBeaverOracle _beaverOracle;

        public YRSimpleOracleBeaver(IYREqualityBeaverOracle beaverOracle)
        {
            _beaverOracle = beaverOracle;
        }
        public override bool CompareContext(IYRContext premise, IYRContext conclusion)
        {
            List<IYRStatement> premiseStatements = new List<IYRStatement>();
            premiseStatements.Add(premise.Statement);
            while (premise != premise.Context)
            {
                premiseStatements.Add(premise.Statement);
                premise = premise.Context;
            }

            List<IYRStatement> conclusionStatements = new List<IYRStatement>();
            conclusionStatements.Add(conclusion.Statement);
            while (conclusion != conclusion.Context)
            {
                conclusionStatements.Add(conclusion.Statement);
                conclusion = conclusion.Context;
            }

            return conclusionStatements.All(x => premiseStatements.All(y => AskOracle(y, x)));
        }

        private bool AskOracle(IYRStatement y, IYRStatement x)
        {
            var result = _beaverOracle.Forsee(y, x).GetAwaiter().GetResult();
            return result;
        }
    }

Please follow these requirements and instructions when writing code: 
Naming Conventions for classed, interfaces and enums: 
ALL interface should always start their name with the Prefix IYR
All classes should always start their name with the Prefix YR
All enums should always start their name with the Prefix YR
All structs should always start their name with the Prefix YR
All records should always start their name with the Prefix YR

The values of enums should always be in ALL-CAPS (but not the name of the enum itself, that is still PascalCase)
every enum should also always, in addition to its other values contain the value NULL = 0 (representing an invalid value and meaning it hasnt been initalized correctly), that helps later in debugging as you can see if an Enum has been initialized even.
In the case, where if we have a value like NONE for our enum, then even that value becomes the first enum value (NONE = 1) (so NULL doesnt mean, like no value, but is a completely seperate value)
Try to think about the general Architecture first and what interfaces we need first.
Use first principal thinking and try to break down the problem into small parts.
We dont have to solve the problem all at once. We can solve it step by step.
Styleguide for C# (refular business logic .cs files) code: 
Optimal and Minimal Code
Never use Comments
Variable names should always be descriptive. Never use abstract symbols or abbreviations for variable names. 

Generally use classes, interfaces for general business logic.
Use enums for finite Enumerations (like day of the week).
Use a record when the primary purpose is to carry data.
Use a struct when objects represents a single, atomic value, has a small memory footprint and will not be "boxed" or "unboxed" frequently.



Each class, no matter how trivial, must origin atleast from one interface that we define (multiple implementations are of course still possible without having to define multiple interfaces).
Each class, interface and enum goes into each own file.
Each namespace represents a distinct module and can be thought of having its own C# project / being its own Nuget-Module
Always think about how to solve the problem by using an OOP approach.
We never want a class to be too large. Classes should definetly be (except VERY rare exceptions) not more than 200 lines of code and Methods (except VERY rare exceptions) shouldnt have more than 40 lines of code
Make all classes and interfaces public for now so we can use them in other libraries
Always answer with the full Code for any class or method that you change, so it can be copied more easily.
Try to have everything that makes sense and is some kind of a computation return a ITask or a IValueTask (both special optimized async classes) such that we always have the possibility to use multithreading in the future.
DONT put 'async' in these methods name, all our methods are basically async.
Also, you DONT need to use ValueTask<T>.AsIValueTask(), you can just make the method an async IValueTask<T> method and return value of type T directly!
